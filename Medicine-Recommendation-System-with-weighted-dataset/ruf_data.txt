def diagnosisGemini(finalSym, predicted_disease):
    """Generates a JSON object listing the primary predicted disease along with other possible diseases(maximum 5),
    descriptions, precautions, workouts, medications, and diets for user interface display in a Health Consultancy project.

    Args:
        finalSym: A list or string of symptoms provided by the user.
        predicted_disease: A specific disease to include in the response, with its description.

    Returns:
        A JSON string with fields for disease, disease_description, precautions, workout, medications, and diets."""

    # Adjusted prompt with inclusion of predicted_disease
    prompt = (
        f"Based on the symptoms provided: {finalSym}, return a JSON object in the following format:\n"
        "{\n"
        "  'possible_diseases': [\n"
        "    { 'name': '" + predicted_disease + "', 'description': 'Description of " + predicted_disease + "' },\n"
        "    { 'name': 'Disease2', 'description': 'Description of Disease2' },\n"
        "    { 'name': 'Disease3', 'description': 'Description of Disease3' },\n"
        "    { 'name': 'Disease4', 'description': 'Description of Disease4' }\n"
        "  ],\n"
        "  'precautions': ['Precaution1', 'Precaution2'],\n"
        "  'workout': ['Suggestion1', 'Suggestion2'],\n"
        "  'medications': ['Medication1', 'Medication2'],\n"
        "  'diets': ['Diet1', 'Diet2']\n"
        "}\n"
        "Make sure that the predicted disease is included along with other diseases that could match the symptoms."
    )

    # Calling the Gemini model
    model = genai.GenerativeModel("gemini-1.5-flash")
    response = model.generate_content(prompt)

    return response.text
